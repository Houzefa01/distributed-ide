\begin{frame}
  \frametitle{CRDT Core: Le Moteur}
  
  Composants cles:
  \begin{itemize}
    \item CRDTIdentifier: ID unique (timestamp, nodeId, position)
    \item CRDTElement: Caractere + ID + tombstone flag
    \item CRDTOperation: INSERT ou DELETE operations
    \item CRDTDocument: Liste ordonnee d'elements
    \item CRDTEngine: Merge + Garbage collection
  \end{itemize}
  
  \vspace{0.5cm}
  
  \centering
  \textbf{57 tests unitaires - Convergence prouvee}
\end{frame}
