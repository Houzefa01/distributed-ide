% SLIDE 6: CRDT CORE
\begin{frame}
  \frametitle{CRDT Core: Le Moteur}
  
  \textbf{5 Composants Cles:}
  
  \begin{enumerate}
    \item \textbf{CRDTIdentifier} - ID unique (timestamp, nodeID, position)
    \item \textbf{CRDTElement} - Caractere + ID + tombstone
    \item \textbf{CRDTOperation} - INSERT/DELETE operations
    \item \textbf{CRDTDocument} - Liste d'elements
    \item \textbf{CRDTEngine} - Merge + GC
  \end{enumerate}
  
  \bigskip
  
  \centering
  
  \Large\textbf{57 Tests Unitaires - Convergence Prouvee}
\end{frame}
