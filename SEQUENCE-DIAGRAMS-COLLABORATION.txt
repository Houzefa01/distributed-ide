\textbf{Points clés :}
\begin{itemize}
    \item ID unique créé à chaque insertion : $(timestamp, nodeId, disambiguator)$
    \item Insertion O(log n) dans SortedMap
    \item Opération enregistrée pour synchronisation
    \item Listeners notifiés pour mise à jour UI
    \item Pas de dépendance globale (offline-ready)
\end{itemize}

$$\text{Replica A timeline:} \quad \text{Insert A} \to \text{Insert B}$$
$$\text{Replica B timeline:} \quad \text{Insert B} \to \text{Insert A}$$
$$\text{Résultat :} \quad "BH" \text{ (identique sur tous les réplicas)}$$

\textbf{Propriétés Critiques :}
\begin{enumerate}
    \item \textbf{Idempotence :} Si déjà présent, skip (ligne 3)
    \item \textbf{Commutativité :} SortedMap tri automatiquement les IDs
    \item \textbf{Causalité :} logicalClock augmente → ordre préservé
    \item \textbf{Soft delete :} Jamais supprimer physiquement
\end{enumerate}

\begin{enumerate}
    \item \textbf{Signaling} : Négociation via serveur central (signaling)

    \item \textbf{ICE} : Échange candidates pour établir tunnel
    \item \textbf{P2P Direct} : Connexion directe entre pairs (sans central)
\end{enumerate}

\textbf{Clé du succès CRDT :}
\begin{itemize}
    \item Alice voit result local "iv" avant ajout Alice
    \item Bob voit result local "vi" avant ajout Bob
    \item Les deux convergent vers même ordre car ordre CRDT détérministe
    \item Pas de version centrale = chacun peut éditer offline
    \item Réseau latence n'importe pas (asynchrone)
\end{itemize}

\section*{Propriétés Mathématiques Appliquées}

\subsection*{Commutativité}
L'ordre d'application des opérations n'affecte pas le résultat final :
$$\forall op_1, op_2 : \text{apply}(op_1, op_2) = \text{apply}(op_2, op_1)$$

\subsection*{Idempotence}
L'application répétée d'une même opération = une seule application :
$$\forall op : \text{apply}(\text{apply}(op)) = \text{apply}(op)$$

\subsection*{Causalité}
L'horloge logique préserve l'ordre des événements au sein d'un nœud :
$$\forall op_i, op_j \text{ du même nœud} : ts_i < ts_j \Rightarrow op_i \text{ avant } op_j$$

\subsection*{Convergence}
Tous les réplicas convergeront au même état malgré la latence réseau :
$$\forall \text{ réplicas } A, B : 
  \lim_{t \to \infty} \text{state}_A(t) = \text{state}_B(t)$$
